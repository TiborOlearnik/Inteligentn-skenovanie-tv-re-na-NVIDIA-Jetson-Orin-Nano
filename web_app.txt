LOG_FILE = os.path.join(BASE_DIR, "auth_log.csv")
os.makedirs(AUTH_DIR, exist_ok=True)

# Cache biometrických vektorov v pamäti: {meno: {"encoding": np.array, "filename": str}}
users_db = {}


def resize_if_needed(bgr_img, max_dim=1280):
    """Zmenší príliš veľké fotky (napr. z mobilu), aby šetril 8 GB GPU RAM na Jetsone."""
    h, w = bgr_img.shape[:2]
    if max(h, w) > max_dim:
        scale = max_dim / float(max(h, w))
        bgr_img = cv2.resize(bgr_img, (int(w * scale), int(h * scale)), interpolation=cv2.INTER_AREA)
    return bgr_img


def load_users_from_disk():
    """Načíta registrovaných používateľov z priečinka web_data/authorized."""
    users_db.clear()
    for fname in sorted(os.listdir(AUTH_DIR)):
        if fname.lower().endswith((".jpg", ".jpeg", ".png")):
            path = os.path.join(AUTH_DIR, fname)
            img = face_recognition.load_image_file(path)
            encs = face_recognition.face_encodings(img)
            if encs:
                name = os.path.splitext(fname)[0]
                users_db[name] = {"encoding": encs[0], "filename": fname}
                print(f"[OK] Načítaný používateľ: {name}")


def log_attempt(status, name, distance):
    """Zapíše výsledok overenia do CSV súboru."""
    file_exists = os.path.isfile(LOG_FILE)
    with open(LOG_FILE, mode="a", newline="", encoding="utf-8") as f:
        writer = csv.writer(f)
        if not file_exists:
            writer.writerow(["Cas", "Stav", "Osoba", "Vzdialenost"])
        writer.writerow([
            datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
            status,
            name,
            f"{distance:.3f}" if distance is not None else "-"
        ])


def read_logs(limit=10):
    """Prečíta posledné záznamy z histórie overení."""
    if not os.path.isfile(LOG_FILE):
        return []
    with open(LOG_FILE, mode="r", encoding="utf-8") as f:
        rows = list(csv.reader(f))
    if len(rows) <= 1:
        return []
    return list(reversed(rows[1:]))[:limit]

        tolerance=0.50,
        message=None,
        success=True,
        result_image=None
    )


@app.route("/user_photo/<filename>")
def user_photo(filename):
    return send_from_directory(AUTH_DIR, filename)


@app.route("/register", methods=["POST"])
def register():
    raw_name = request.form.get("name", "").strip()
    safe_name = secure_filename(raw_name)
    file = request.files.get("photo")

    if not safe_name or not file:
        return render_template_string(
            HTML_TEMPLATE, users=users_db, logs=read_logs(), tolerance=0.50,
            message="Zadaj platné meno a vyber fotografiu.", success=False, result_image=None
        )

    file_bytes = np.frombuffer(file.read(), np.uint8)
    bgr_img = cv2.imdecode(file_bytes, cv2.IMREAD_COLOR)
    if bgr_img is None:
        return render_template_string(
            HTML_TEMPLATE, users=users_db, logs=read_logs(), tolerance=0.50,
            message="Súbor sa nepodarilo načítať ako obrázok.", success=False, result_image=None
        )

    bgr_img = resize_if_needed(bgr_img)
    rgb_img = cv2.cvtColor(bgr_img, cv2.COLOR_BGR2RGB)

    locations = face_recognition.face_locations(rgb_img, model="cnn")
    if len(locations) != 1:
        msg = (
            "Na fotografii sa nenašla žiadna tvár."
            if len(locations) == 0
            else f"Na fotografii sa našlo viac tvárí ({len(locations)}). Nahraj fotku iba jednej osoby."
        )
        return render_template_string(
            HTML_TEMPLATE, users=users_db, logs=read_logs(), tolerance=0.50,
            message=msg, success=False, result_image=None
        )

    encodings = face_recognition.face_encodings(rgb_img, locations)
    filename = f"{safe_name}.jpg"
    save_path = os.path.join(AUTH_DIR, filename)
    cv2.imwrite(save_path, bgr_img)

    users_db[safe_name] = {"encoding": encodings[0], "filename": filename}
    return render_template_string(
        HTML_TEMPLATE, users=users_db, logs=read_logs(), tolerance=0.50,
        message=f"Používateľ '{safe_name}' bol úspešne zaregistrovaný.", success=True, result_image=None
    )


@app.route("/verify", methods=["POST"])
def verify():
    tolerance = float(request.form.get("tolerance", 0.50))
    file = request.files.get("photo")

    if not users_db:
        return render_template_string(
            HTML_TEMPLATE, users=users_db, logs=read_logs(), tolerance=tolerance,
            message="Databáza je prázdna. Najprv zaregistruj aspoň jedného používateľa.",
            success=False, result_image=None
        )

    file_bytes = np.frombuffer(file.read(), np.uint8)
    bgr_img = cv2.imdecode(file_bytes, cv2.IMREAD_COLOR)
    if bgr_img is None:
        return render_template_string(
            HTML_TEMPLATE, users=users_db, logs=read_logs(), tolerance=tolerance,
            message="Neplatný formát obrázka.", success=False, result_image=None
        )

    bgr_img = resize_if_needed(bgr_img)
    rgb_img = cv2.cvtColor(bgr_img, cv2.COLOR_BGR2RGB)

    # Detekcia na CUDA GPU
    locations = face_recognition.face_locations(rgb_img, model="cnn")
    encodings = face_recognition.face_encodings(rgb_img, locations)

    if not locations:
        log_attempt("BEZ_TVARE", "-", None)
        return render_template_string(
            HTML_TEMPLATE, users=users_db, logs=read_logs(), tolerance=tolerance,
            message="Na nahranej fotografii sa nenašla žiadna tvár.", success=False, result_image=None
        )

    known_names = list(users_db.keys())
    known_encodings = [users_db[n]["encoding"] for n in known_names]

    any_authorized = False
    summary_parts = []

    for (top, right, bottom, left), enc in zip(locations, encodings):
        distances = face_recognition.face_distance(known_encodings, enc)
        best_idx = int(np.argmin(distances))
        best_dist = float(distances[best_idx])

        if best_dist <= tolerance:
            matched_name = known_names[best_idx]
            label = f"AUTORIZOVANY: {matched_name} ({best_dist:.2f})"
            color = (0, 255, 0)
            any_authorized = True
            summary_parts.append(f"{matched_name} (dist: {best_dist:.2f})")
            log_attempt("AUTORIZOVANY", matched_name, best_dist)
        else:
            label = f"NEAUTORIZOVANY ({best_dist:.2f})"
            color = (0, 0, 255)
            summary_parts.append(f"Neznáma osoba (dist: {best_dist:.2f})")
            log_attempt("NEAUTORIZOVANY", "Neznámy", best_dist)

        cv2.rectangle(bgr_img, (left, top), (right, bottom), color, 2)
        cv2.putText(bgr_img, label, (left, max(25, top - 10)),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.65, color, 2)

    _, buffer = cv2.imencode(".jpg", bgr_img)
    result_b64 = base64.b64encode(buffer).decode("utf-8")

    msg = "Výsledok overenia: " + ", ".join(summary_parts)
    return render_template_string(
        HTML_TEMPLATE, users=users_db, logs=read_logs(), tolerance=tolerance,
        message=msg, success=any_authorized, result_image=result_b64
    )


@app.route("/delete/<name>", methods=["POST"])
def delete_user(name):
    if name in users_db:
        filepath = os.path.join(AUTH_DIR, users_db[name]["filename"])
        if os.path.exists(filepath):
            os.remove(filepath)
        del users_db[name]
    return redirect(url_for("index"))
